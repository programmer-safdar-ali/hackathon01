# capstone_robot/nodes/perception_node.py
import rclpy
from rclpy.node import Node
from vision_msgs.msg import Detection2DArray

class PerceptionNode(Node):
    def __init__(self):
        super().__init__('perception_node')
        self.detection_sub = self.create_subscription(
            Detection2DArray,
            '/detectnet/detections',
            self.detection_callback,
            10
        )
        self.get_logger().info('Perception node started')

    def detection_callback(self, msg: Detection2DArray):
        for detection in msg.detections:
            self.get_logger().info(
                f'Detected: {detection.results[0].hypothesis.class_id} '
                f'confidence={detection.results[0].hypothesis.score:.2f}'
            )

## Step 5: Build the Voice Command Pipeline

**References**: [Module 4 Ch.2 — Whisper Voice](/docs/module-4-vla/whisper-voice), [Module 4 Ch.3 — LLM Planning](/docs/module-4-vla/llm-planning)

```python
# capstone_robot/nodes/voice_node.py
import rclpy
from rclpy.node import Node
from std_msgs.msg import String
import whisper
import sounddevice as sd
import numpy as np

class VoiceCommandNode(Node):
    def __init__(self):
        super().__init__('voice_command_node')
        self.model = whisper.load_model('base')
        self.command_pub = self.create_publisher(String, '/voice_command', 10)
        self.sample_rate = 16000
        self.get_logger().info('Voice node ready — say a command')

    def record_and_transcribe(self, duration: float = 3.0) -> str:
        audio = sd.rec(
            int(duration * self.sample_rate),
            samplerate=self.sample_rate,
            channels=1,
            dtype='float32'
        )
        sd.wait()
        audio_flat = audio.flatten()
        result = self.model.transcribe(audio_flat)
        return result['text'].strip()

    def publish_command(self, text: str):
        msg = String()
        msg.data = text
        self.command_pub.publish(msg)
        self.get_logger().info(f'Voice command: "{text}"')
```

**Checkpoint ✓**: Speaking "pick up the red cube" publishes to `/voice_command` topic.

## Step 6: Implement LLM Task Planning

**References**: [Module 4 Ch.3 — LLM Planning](/docs/module-4-vla/llm-planning)

```python
# capstone_robot/services/task_planner.py
from openai import OpenAI

client = OpenAI()

SYSTEM_PROMPT = """You are a robot task planner. Convert natural language commands
into structured robot action sequences. Available actions:
- NAVIGATE_TO(x, y) — move base to position
- DETECT_OBJECT(class_name) — run perception for object
- PICK_UP(object_id) — grasp detected object
- PLACE_AT(x, y, z) — place held object at location
- SPEAK(message) — say message via TTS

Return a JSON array of action steps only."""

def plan_task(voice_command: str) -> list[dict]:
    response = client.chat.completions.create(
        model='gpt-4o-mini',
        messages=[
            {'role': 'system', 'content': SYSTEM_PROMPT},
            {'role': 'user', 'content': voice_command},
        ],
        response_format={'type': 'json_object'},
    )
    import json
    return json.loads(response.choices[0].message.content)['actions']
```

**Checkpoint ✓**: `plan_task("pick up the red cube and put it on the table")` returns a valid action sequence JSON.

## Step 7: Implement the Motion Planning Action Server

**References**: [Module 1 Ch.3 — Services & Actions](/docs/module-1-ros2/services-actions)

```python
# capstone_robot/actions/manipulation_server.py
import rclpy
from rclpy.node import Node
from rclpy.action import ActionServer
from capstone_interfaces.action import ManipulateObject

class ManipulationActionServer(Node):
    def __init__(self):
        super().__init__('manipulation_server')
        self._action_server = ActionServer(
            self,
            ManipulateObject,
            'manipulate_object',
            self.execute_callback
        )

    async def execute_callback(self, goal_handle):
        self.get_logger().info(f'Executing: {goal_handle.request.action_type}')
        feedback = ManipulateObject.Feedback()

        # Simulate action phases
        for progress in [25, 50, 75, 100]:
            feedback.progress_percent = progress
            goal_handle.publish_feedback(feedback)
            await rclpy.sleep(0.5)

        goal_handle.succeed()
        result = ManipulateObject.Result()
        result.success = True
        return result
```

**Checkpoint ✓**: `ros2 action list` shows `/manipulate_object`. Sending a goal receives feedback updates.

## Step 8: Integrate Sim-to-Real Transfer

**References**: [Module 3 Ch.4 — Sim-to-Real](/docs/module-3-isaac/sim-to-real)

Apply domain randomization before real-hardware deployment:

```python
# capstone_robot/training/domain_randomizer.py
import random

class DomainRandomizer:
    """Randomize simulation parameters to improve real-world transfer."""

    def randomize_lighting(self, world_config: dict) -> dict:
        world_config['ambient_light'] = random.uniform(0.3, 1.0)
        world_config['light_direction'] = [
            random.uniform(-1, 1),
            random.uniform(-1, 1),
            random.uniform(0.5, 1.0),
        ]
        return world_config

    def randomize_object_positions(self, objects: list, radius: float = 0.1) -> list:
        return [
            {**obj, 'x': obj['x'] + random.uniform(-radius, radius),
                    'y': obj['y'] + random.uniform(-radius, radius)}
            for obj in objects
        ]

    def randomize_camera_noise(self, image_array):
        noise = np.random.normal(0, 0.02, image_array.shape)
        return np.clip(image_array + noise, 0, 1)
```

**Checkpoint ✓**: 100 randomized simulation episodes complete without crashes. Transfer policy tested on at least 3 object positions.

import pytest
import rclpy
from capstone_robot.coordinator import RobotCoordinator

@pytest.mark.integration
def test_pick_and_place_pipeline():
    """Test complete pick-and-place task from voice command to execution."""

    # Simulate voice command
